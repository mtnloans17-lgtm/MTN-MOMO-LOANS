/** * Screen State Transition Manager * Handles showing and hiding the single-page view panels * @param {string} screenId - The ID of the HTML element container to show */ function showScreen(screenId) { document.querySelectorAll('.screen-container').forEach(screen => { screen.classList.add('hidden'); }); const targetScreen = document.getElementById(screenId); if (targetScreen) { targetScreen.classList.remove('hidden'); } window.scrollTo(0, 0); } /** * Form Submit Interceptor & API Event Handler * Binds to the loan layout configuration form element */ document.addEventListener('DOMContentLoaded', () => { const loanForm = document.getElementById('loanForm'); if (loanForm) { loanForm.addEventListener('submit', async function(event) { event.preventDefault(); // Extract core actionable DOM items const submitButton = this.querySelector('button[type="submit"]'); const fullNameInput = this.querySelector('input[placeholder="Enter your full name"]'); const phoneInput = this.querySelector('input[placeholder="07XXXXXXXX"]'); const amountSelect = this.querySelector('select:nth-of-type(1)'); const periodSelect = this.querySelector('select:nth-of-type(2)'); // Validate form field nodes exist before parsing metrics if (!fullNameInput || !phoneInput || !amountSelect || !periodSelect) { alert('Structural Error: Form input components could not be evaluated.'); return; } // Map data structure variables const payload = { fullName: fullNameInput.value, phoneNumber: phoneInput.value, amount: parseFloat(amountSelect.value), periodMonths: parseInt(periodSelect.value) }; try { // UI Visual State Optimization - Disable double tap submittals if (submitButton) { submitButton.disabled = true; submitButton.innerHTML = ' Processing...'; } // Dispatches asynchronous telemetry out to the Node Express server engine const response = await fetch('http://localhost:3000/api/loans/apply', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload) }); const jsonResult = await response.json(); if (response.ok && jsonResult.success) { // Inject database/backend validated variables directly into confirmation fields const receiptCard = document.querySelector('.receipt-card'); if (receiptCard) { receiptCard.innerHTML = `
Loan Amount: UGX ${jsonResult.data.amount.toLocaleString()}
Repayment Period: ${jsonResult.data.periodMonths} Months
Mobile Number: ${jsonResult.data.phoneNumber}
Application Status: ${jsonResult.data.status}
`; } // Route user to the final confirmation receipt wrap interface showScreen('screen-success'); } else { // Display validation context flags sent back from backend endpoints alert('Application Error: ' + (jsonResult.message || 'System validation rejected metrics parameters.')); } } catch (error) { console.error('API Server connection state failed:', error); alert('Communication Loss: Connection to local endpoint on port 3000 refused. Please ensure your backend engine is active.'); } finally { // Revert visual configurations back to interactive default parameters if (submitButton) { submitButton.disabled = false; submitButton.innerHTML = 'Continue '; } } }); } });
